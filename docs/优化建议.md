```python
def verify_cookie(cookies):
    """验证Cookie是否完整"""
    # 1. 检查Cookie数量
    if len(cookies) < 20:
        return False, "Cookie数量太少"
    
    # 2. 检查关键Cookie
    required_cookies = ['MSPOK', 'MSPRequ', 'MSCC']
    for name in required_cookies:
        if not any(c['name'] == name for c in cookies):
            return False, f"缺少关键Cookie: {name}"
    
    # 3. 测试Cookie是否有效
    # 访问一个需要登录的页面，看是否会重定向到登录页面
    
    return True, "Cookie完整"
```

#### 5. 添加Cookie过期检测

**问题**：

**解决方案**：
在自动登录前，检查Cookie是否过期：

```python
def is_cookie_expired(cookie_file):
    """检查Cookie是否过期"""
    metadata = cookie_file.get('metadata', {})
    created_at = metadata.get('created_at')
    
    if not created_at:
        return False, "无法确定Cookie创建时间"
    
    # Cookie有效期通常是24小时
    created_time = datetime.strptime(created_at, "%Y-%m-%d %H:%M:%S")
    now = datetime.now()
    age = (now - created_time).total_seconds() / 3600  # 小时
    
    if age > 24:
        return True, f"Cookie已过期（{age:.1f}小时）"
    
    return False, f"Cookie有效（{age:.1f}小时）"
```

**实现步骤**：
1. 创建Cookie过期检测函数
2. 在自动登录前调用检测函数
3. 如果过期，提示用户重新获取

**预期效果**：

**问题**：

**解决方案**：
自动检测并重新获取Cookie：

```python
def auto_login_with_retry(email, password, device_code):
    """自动登录，如果Cookie无效则重新获取"""
    # 1. 尝试使用现有Cookie登录
    success, error = try_login_with_cookie(email, password, device_code)
    
    if success:
        return True, "登录成功"
    
    # 2. 如果失败，检查是否是Cookie问题
    if error == "cookie_invalid" or error == "cookie_expired":
        print("[信息] Cookie无效，正在重新获取...")
        
        # 3. 重新获取Cookie
        success, error = get_cookie(email, password)
        if not success:
            return False, f"重新获取Cookie失败: {error}"
        
        # 4. 使用新Cookie重新登录
        success, error = try_login_with_cookie(email, password, device_code)
        return success, error
    
    return False, error
```

```python
class AccountManager:
    def __init__(self):
        self.accounts = self.load_accounts()
    
    def load_accounts(self):
        """加载保存的账号列表"""
        if os.path.exists('accounts.json'):
            with open('accounts.json', 'r') as f:
                return json.load(f)
        return []
    
    def save_accounts(self):
        """保存账号列表"""
        with open('accounts.json', 'w') as f:
            json.dump(self.accounts, f, indent=2)
    
    def add_account(self, email, password):
        """添加账号"""
        self.accounts.append({
            'email': email,
            'password': password,
            'cookie_file': f'cookies/{email.replace("@", "_")}.json',
            'last_login': None
        })
        self.save_accounts()
    
    def remove_account(self, email):
        """删除账号"""
        self.accounts = [a for a in self.accounts if a['email'] != email]
        self.save_accounts()
    
    def get_account(self, email):
        """获取账号信息"""
        for account in self.accounts:
            if account['email'] == email:
                return account
        return None
    
    def list_accounts(self):
        """列出所有账号"""
        return self.accounts
```
